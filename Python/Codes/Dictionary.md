        entry.insert(0, word)
        show_definition(word)
        listbox.place_forget()


def show_definition(word=None):
    if word is None:
        word = entry.get().strip().lower()

    result_text.delete("1.0", tk.END)
    if word in words_dict:
        result_text.insert(tk.END, f"Word :{word}\n\nDefinition :\n{words_dict[word]}")
    else:
        result_text.insert(tk.END, f"{word} not found in dictionary.")

    def clear():
        entry.delete(0, "end")
        listbox.delete(0, "end")
        listbox.selection_clear(0, "end")
        result_text.delete("1.0", "end")

    entry.focus_set()
